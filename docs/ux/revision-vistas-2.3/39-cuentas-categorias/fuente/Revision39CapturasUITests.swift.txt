//  Revision39CapturasUITests.swift — TEMPORAL, no se publica.
//  Conductor de capturas de la revisión por vistas 2.3, zona 39 (cuentas, tarjetas, categorías, etiquetas y divisas).
//  `TEST_RUNNER_REV_OUT=<carpeta>` escribe ahí los PNG; `TEST_RUNNER_REV_AX3=1` lanza con texto grande (AX3);
//  `TEST_RUNNER_REV_PREFIX` antepone un prefijo. Cada caso tolera fallos: captura lo que alcance.
//

import XCTest

final class Revision39CapturasUITests: XCTestCase {

    private static let ax3 = ["-UIPreferredContentSizeCategoryName", "UICTContentSizeCategoryAccessibilityXL"]
    private var prefix: String { ProcessInfo.processInfo.environment["REV_PREFIX"] ?? "" }

    override func setUp() {
        super.setUp()
        continueAfterFailure = true
    }

    // MARK: - Utilidades

    private func cap(_ name: String, wait: TimeInterval = 0.8) {
        Thread.sleep(forTimeInterval: wait)
        guard let dir = ProcessInfo.processInfo.environment["REV_OUT"], !dir.isEmpty else { return }
        let data = XCUIScreen.main.screenshot().pngRepresentation
        let url = URL(fileURLWithPath: dir).appendingPathComponent(prefix + name + ".png")
        do { try data.write(to: url) } catch { print("REV: no se pudo escribir \(url): \(error)") }
    }

    private func launch(seed: String?, pro: Bool = true, extra: [String] = [], forcePEN: Bool = true) -> XCUIApplication {
        let app = XCUIApplication()
        let large = ProcessInfo.processInfo.environment["REV_AX3"] == "1" ? Self.ax3 : []
        let currency = forcePEN ? ["-defaultCurrencyCode", "PEN"] : ["-AppleLocale", "es_PE"]
        app.launchForUITest(pro: pro, seed: seed, extraArguments: extra + large + currency)
        _ = app.waitForUITestReady()
        Thread.sleep(forTimeInterval: 3)
        return app
    }

    @discardableResult
    private func tapIf(_ el: XCUIElement, timeout: TimeInterval = 8) -> Bool {
        guard el.waitForExistence(timeout: timeout) else { print("REV: no apareció \(el)"); return false }
        el.tap()
        return true
    }

    private func openSettingsRow(_ app: XCUIApplication, _ id: String) {
        tapIf(app.buttons["profile_avatar"], timeout: 20)
        let row = app.buttons[id].firstMatch
        var n = 0
        while !(row.exists && row.isHittable) && n < 8 { app.swipeUp(); n += 1 }
        tapIf(row, timeout: 6)
    }

    private func scrollCaps(_ app: XCUIApplication, _ base: String, count: Int) {
        for i in 1...count {
            app.swipeUp()
            cap("\(base)-\(i)", wait: 0.6)
        }
    }

    private func back(_ app: XCUIApplication) {
        let b = app.navigationBars.buttons.firstMatch
        if b.waitForExistence(timeout: 3) { b.tap() }
        Thread.sleep(forTimeInterval: 0.8)
    }
