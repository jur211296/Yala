    // MARK: - Utilidades

    private func cap(_ name: String, wait: TimeInterval = 1.0) {
        Thread.sleep(forTimeInterval: wait)
        guard let dir = ProcessInfo.processInfo.environment["REV_OUT"], !dir.isEmpty else { return }
        let data = XCUIScreen.main.screenshot().pngRepresentation
        let url = URL(fileURLWithPath: dir).appendingPathComponent(prefix + name + ".png")
        do { try data.write(to: url) } catch { print("REV: no se pudo escribir \(url): \(error)") }
    }

    private func launch(seed: String? = "minimal", pro: Bool = false, configure: (XCUIApplication) -> Void) -> XCUIApplication {
        let app = XCUIApplication()
        configure(app)
        _ = app.waitForUITestReady()
        Thread.sleep(forTimeInterval: 2.5)
        return app
    }

    private var large: [String] { ProcessInfo.processInfo.environment["REV_AX3"] == "1" ? Self.ax3 : [] }

    private func any(_ app: XCUIApplication, _ id: String) -> XCUIElement { app.descendants(matching: .any)[id].firstMatch }

    @discardableResult
    private func tapIf(_ el: XCUIElement, timeout: TimeInterval = 8) -> Bool {
        guard el.waitForExistence(timeout: timeout) else { print("REV: no apareció \(el)"); return false }
        el.tap()
        return true
    }

    @discardableResult
    private func scrollTo(_ app: XCUIApplication, _ id: String) -> XCUIElement {
        let el = app.descendants(matching: .any)[id].firstMatch
        var n = 0
        while !(el.exists && el.isHittable) && n < 12 { app.swipeUp(); n += 1; Thread.sleep(forTimeInterval: 0.3) }
        return el
    }

    private func openStorage(_ app: XCUIApplication, capRow: String? = nil) {
        app.openProfile()
        let row = scrollTo(app, "storage_settings_row")
        if let capRow { cap(capRow, wait: 0.8) }
        tapIf(row, timeout: 10)
        Thread.sleep(forTimeInterval: 1.5)
    }

    private func scrollCaps(_ app: XCUIApplication, _ base: String, count: Int) {
        for i in 1...count {
            app.swipeUp()
            cap("\(base)-\(i)", wait: 0.7)
        }
    }

    // MARK: - Reposo, para medir CPU desde fuera (ps sobre el proceso de la app)

    /// Marca en un fichero cuándo empieza y acaba cada tramo de 60 s; el host muestrea el tiempo de CPU del proceso.
    private func mark(_ text: String) {
        guard let dir = ProcessInfo.processInfo.environment["REV_OUT"], !dir.isEmpty else { return }
        let url = URL(fileURLWithPath: dir).appendingPathComponent("perf-marcas.txt")
        let line = "\(Date().timeIntervalSince1970) \(text)\n"
        if let h = try? FileHandle(forWritingTo: url) { h.seekToEndOfFile(); h.write(line.data(using: .utf8)!); h.closeFile() }
        else { try? line.write(to: url, atomically: true, encoding: .utf8) }
    }
