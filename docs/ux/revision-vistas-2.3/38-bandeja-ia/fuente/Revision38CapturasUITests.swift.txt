import XCTest

final class Revision38CapturasUITests: XCTestCase {

    private static let ax3 = ["-UIPreferredContentSizeCategoryName", "UICTContentSizeCategoryAccessibilityXL"]
    private var prefix: String { ProcessInfo.processInfo.environment["REV_PREFIX"] ?? "" }

    override func setUp() {
        super.setUp()
        continueAfterFailure = true
    }

    // MARK: - Utilidades

    private func cap(_ name: String, wait: TimeInterval = 0.8) {
        Thread.sleep(forTimeInterval: wait)
        guard let dir = ProcessInfo.processInfo.environment["REV_OUT"], !dir.isEmpty else { return }
        let data = XCUIScreen.main.screenshot().pngRepresentation
        let url = URL(fileURLWithPath: dir).appendingPathComponent(prefix + name + ".png")
        do { try data.write(to: url) } catch { print("REV: no se pudo escribir \(url): \(error)") }
    }

    private func launch(_ extra: [String], seed: String? = "grupos", pro: Bool = false) -> XCUIApplication {
        let app = XCUIApplication()
        let large = ProcessInfo.processInfo.environment["REV_AX3"] == "1" ? Self.ax3 : []
        app.launchForUITest(pro: pro, seed: seed, extraArguments: extra + large + ["-defaultCurrencyCode", "PEN"])
        _ = app.waitForUITestReady()
        Thread.sleep(forTimeInterval: 3)
        return app
    }

    @discardableResult
    private func tapIf(_ el: XCUIElement, timeout: TimeInterval = 8) -> Bool {
        guard el.waitForExistence(timeout: timeout) else { print("REV: no apareció \(el)"); return false }
        el.tap()
        return true
    }

    private func openInbox(_ app: XCUIApplication) {
        tapIf(app.buttons["panel_inbox_button"], timeout: 20)
        _ = app.descendants(matching: .any)["inbox_explainer"].waitForExistence(timeout: 8)
    }

    private func openSettings(_ app: XCUIApplication) {
        tapIf(app.buttons["profile_avatar"], timeout: 20)
        _ = app.buttons["profile_accounts"].waitForExistence(timeout: 10)
    }

    private func scrollToRow(_ app: XCUIApplication, _ el: XCUIElement, max: Int = 12) {
        var n = 0
        while !(el.exists && el.isHittable) && n < max { app.swipeUp(); n += 1 }
    }
