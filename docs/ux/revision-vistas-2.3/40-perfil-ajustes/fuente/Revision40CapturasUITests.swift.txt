//  Revision40CapturasUITests.swift — TEMPORAL, no se publica.
//  Conductor de capturas de la revisión por vistas 2.3, zona 40 (Perfil, Más, Ajustes generales, Pro y ayuda).
//  `TEST_RUNNER_REV_OUT=<carpeta>` escribe ahí los PNG; `TEST_RUNNER_REV_AX3=1` lanza con texto grande (AX3);
//  `TEST_RUNNER_REV_PREFIX` antepone un prefijo. Cada caso tolera fallos: captura lo que alcance.

import XCTest

final class Revision40CapturasUITests: XCTestCase {

    private static let ax3 = ["-UIPreferredContentSizeCategoryName", "UICTContentSizeCategoryAccessibilityXL"]
    private var prefix: String { ProcessInfo.processInfo.environment["REV_PREFIX"] ?? "" }

    override func setUp() {
        super.setUp()
        continueAfterFailure = true
    }

    // MARK: - Utilidades

    private func cap(_ name: String, wait: TimeInterval = 0.8) {
        Thread.sleep(forTimeInterval: wait)
        guard let dir = ProcessInfo.processInfo.environment["REV_OUT"], !dir.isEmpty else { return }
        let data = XCUIScreen.main.screenshot().pngRepresentation
        let url = URL(fileURLWithPath: dir).appendingPathComponent(prefix + name + ".png")
        do { try data.write(to: url) } catch { print("REV: no se pudo escribir \(url): \(error)") }
    }

    private func launch(seed: String? = "realista", pro: Bool = false, extra: [String] = [],
                        fakePrices: Bool = false, fakePurchase: Bool = false, proTrialDays: Int? = nil,
                        notificationsDenied: Bool = false, cloudSession: Bool = false) -> XCUIApplication {
        let app = XCUIApplication()
        let large = ProcessInfo.processInfo.environment["REV_AX3"] == "1" ? Self.ax3 : []
        app.launchForUITest(pro: pro, seed: seed, cloudSession: cloudSession,
                            notificationsDenied: notificationsDenied, fakePrices: fakePrices,
                            fakePurchase: fakePurchase, proTrialDays: proTrialDays,
                            extraArguments: extra + large + ["-defaultCurrencyCode", "PEN"])
        _ = app.waitForUITestReady()
        Thread.sleep(forTimeInterval: 3)
        return app
    }

    @discardableResult
    private func tapIf(_ el: XCUIElement, timeout: TimeInterval = 8) -> Bool {
        guard el.waitForExistence(timeout: timeout) else { print("REV: no apareció \(el)"); return false }
        el.tap()
        return true
    }

    private func openSettings(_ app: XCUIApplication) {
        tapIf(app.buttons["profile_avatar"].firstMatch, timeout: 20)
        Thread.sleep(forTimeInterval: 1.2)
    }

    @discardableResult
    private func scrollTo(_ app: XCUIApplication, _ el: XCUIElement, max: Int = 10) -> Bool {
        var n = 0
        while !(el.exists && el.isHittable) && n < max { app.swipeUp(); n += 1 }
        return el.exists
    }

    private func openSettingsRow(_ app: XCUIApplication, _ id: String) {
        openSettings(app)
        let row = app.buttons[id].firstMatch
        scrollTo(app, row, max: 8)
        tapIf(row, timeout: 6)
        Thread.sleep(forTimeInterval: 1.0)
    }

    private func openSettingsRowByLabel(_ app: XCUIApplication, _ label: String) {
        openSettings(app)
        let row = app.buttons.containing(NSPredicate(format: "label BEGINSWITH %@", label)).firstMatch
        scrollTo(app, row, max: 8)
        tapIf(row, timeout: 6)
        Thread.sleep(forTimeInterval: 1.0)
    }

    private func scrollCaps(_ app: XCUIApplication, _ base: String, count: Int) {
        for i in 1...count {
            app.swipeUp()
            cap("\(base)-\(i)", wait: 0.6)
        }
    }

    private func label(_ text: String) -> NSPredicate {
        NSPredicate(format: "label CONTAINS[c] %@", text)
    }
