    1. Crear SystemDefaults.swift con:
       ```swift
       enum SystemDefaults {
           static func suggestedCurrency() -> CurrencyCode {
               let region = Locale.current.region?.identifier ?? ""
               switch region {
               case "PE": return .pen
               case "MX": return .mxn
               case "CO": return .cop
               case "BR": return .brl
               case "ES", "FR", "DE", "IT", "PT": return .eur
               case "GB": return .gbp
               default: return .usd
               }
           }

           static func suggestedFirstDayOfWeek() -> Int {
               // 1 = Sunday, 2 = Monday
               let region = Locale.current.region?.identifier ?? ""
               // USA, Canada, Japan use Sunday
               return ["US", "CA", "JP"].contains(region) ? 1 : 2
           }

           static var systemLanguageCode: String {
               let preferred = Locale.preferredLanguages.first ?? "en"
               let code = Locale(identifier: preferred).language.languageCode?.identifier ?? "en"
               let supported = ["es", "en", "fr", "de", "it", "pt"]
               return supported.contains(code) ? code : "en"
           }
       }
       ```
    2. Verificar que compila y las funciones devuelven valores correctos

       ```swift
       enum OnboardingState {
           static let hasCompletedKey = "hasCompletedOnboarding"
           static let userNameKey = "userName"

           static var hasCompleted: Bool {
               UserDefaults.standard.bool(forKey: hasCompletedKey)
           }

           static func markCompleted() {
               UserDefaults.standard.set(true, forKey: hasCompletedKey)
           }

           static var userName: String? {
               get { UserDefaults.standard.string(forKey: userNameKey) }
               set { UserDefaults.standard.set(newValue, forKey: userNameKey) }
           }
       }
       ```
